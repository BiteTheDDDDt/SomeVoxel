cmake_minimum_required(VERSION 3.24)
project(SomeVoxel LANGUAGES CXX)

set(SOMEVOXEL_REFINEMENT 4 CACHE STRING "Subdivisions of each 18 cm layout cell")
set(SOMEVOXEL_EXPERIMENTAL_SIMULATION_HZ "" CACHE STRING "Override the experimental simulation update rate")
set(SOMEVOXEL_EXPERIMENTAL_WATER_HZ "" CACHE STRING "Override the experimental bulk water update rate")
option(SOMEVOXEL_WIDE_GRID "Use wide grid metadata for profiling" OFF)
option(SOMEVOXEL_BUILD_GODOT "Build the Godot GDExtension" OFF)
option(SOMEVOXEL_BUILD_EXPERIMENTS "Build simulation experiments" OFF)
option(SOMEVOXEL_EXPERIMENTAL_WATER_SWEEP_PHASE "Advance sweep orientation on bulk-water ticks" OFF)
set(SOMEVOXEL_PARALLEL_WATER_WORKERS "" CACHE STRING "Workers for ordinary water-head rebuilds, including the caller")
set(SOMEVOXEL_PARALLEL_RENDER_WORKERS "" CACHE STRING "Workers for sand render preparation, including the caller")

add_library(somevoxel_core STATIC core/world.cpp)
add_library(SomeVoxel::Core ALIAS somevoxel_core)
target_compile_features(somevoxel_core PUBLIC cxx_std_17)
target_include_directories(somevoxel_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_definitions(somevoxel_core PUBLIC SOMEVOXEL_REFINEMENT=${SOMEVOXEL_REFINEMENT})
foreach(rate IN ITEMS SOMEVOXEL_EXPERIMENTAL_SIMULATION_HZ SOMEVOXEL_EXPERIMENTAL_WATER_HZ)
    if(NOT "${${rate}}" STREQUAL "")
        target_compile_definitions(somevoxel_core PUBLIC "${rate}=${${rate}}")
    endif()
endforeach()
if(SOMEVOXEL_EXPERIMENTAL_WATER_SWEEP_PHASE)
    target_compile_definitions(somevoxel_core PUBLIC SOMEVOXEL_EXPERIMENTAL_WATER_SWEEP_PHASE)
endif()
foreach(workers IN ITEMS SOMEVOXEL_PARALLEL_WATER_WORKERS SOMEVOXEL_PARALLEL_RENDER_WORKERS)
    if(NOT "${${workers}}" STREQUAL "")
        if(NOT "${${workers}}" MATCHES "^([1-9]|1[0-6])$")
            message(FATAL_ERROR "${workers} must be in 1..16")
        endif()
        find_package(Threads REQUIRED)
        target_compile_definitions(somevoxel_core PUBLIC "${workers}=${${workers}}")
        target_link_libraries(somevoxel_core PUBLIC Threads::Threads)
    endif()
endforeach()
set_target_properties(somevoxel_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(SOMEVOXEL_WIDE_GRID)
    target_compile_definitions(somevoxel_core PUBLIC SOMEVOXEL_WIDE_GRID)
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(somevoxel_smoke tests/world_smoke.cpp)
    target_link_libraries(somevoxel_smoke PRIVATE SomeVoxel::Core)
    add_test(NAME somevoxel_world COMMAND somevoxel_smoke)
endif()

if(SOMEVOXEL_BUILD_GODOT)
    add_subdirectory(godot)
endif()

if(SOMEVOXEL_BUILD_EXPERIMENTS)
    find_package(Threads REQUIRED)
    add_executable(somevoxel_parallel_tiles experiments/parallel_tiles.cpp)
    target_compile_features(somevoxel_parallel_tiles PRIVATE cxx_std_17)
    target_link_libraries(somevoxel_parallel_tiles PRIVATE Threads::Threads)
    add_executable(somevoxel_rate_bench experiments/simulation_rate.cpp)
    target_link_libraries(somevoxel_rate_bench PRIVATE SomeVoxel::Core)
endif()
